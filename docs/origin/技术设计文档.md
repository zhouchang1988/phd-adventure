# 《谢凡的奇幻冒险2.0》技术设计文档

## 1. 技术选型

### 1.1 核心技术栈

| 层级 | 技术 | 说明 |
|------|------|------|
| 前端框架 | Next.js 14 (App Router) | React框架，SSR/SSG支持 |
| 语言 | TypeScript | 类型安全，更好的开发体验 |
| UI框架 | Tailwind CSS v4 | 实用优先的CSS框架，零运行时 |
| 组件库 | Shadcn/ui | 可复制粘贴的组件，完全可控 |
| 动画 | Framer Motion + Tailwind Animate | 声明式动画 |
| 存储 | localStorage | 本地存档 |
| 构建 | Turbopack (Next.js内置) | 快速构建 |
| 部署 | Vercel | 零配置部署 |

### 1.2 为什么选择这个方案

**为什么Next.js而不是原生HTML**：
- TypeScript类型安全，减少运行时错误
- 组件化架构，更好的代码复用
- 内置优化（图片、字体、代码分割）
- 丰富的生态系统

**为什么Tailwind CSS而不是纯CSS**：
- 零运行时JavaScript开销
- 12ms构建500组件
- 与Shadcn/ui完美集成
- 主题切换通过CSS变量实现

**为什么不用游戏引擎**：
- **Twine**: 太重，学习成本高，样式定制困难
- **Ren'Py**: Python环境依赖，不适合Web
- **Unity**: 杀鸡用牛刀，2D文字游戏不需要
- **Next.js**: 最灵活，最容易控制，玩家无需安装

### 1.3 可选增强

| 需求 | 方案 | 是否必须 |
|------|------|---------|
| 打字机效果 | 自实现 (useState + setInterval) | 可选 |
| 粒子效果 | Framer Motion | 可选 |
| 音频 | Howler.js | 可选 |
| 存档导出 | FileSaver.js | 可选 |

## 4. 游戏引擎设计

### 4.1 游戏引擎Hook

```typescript
// src/hooks/useGameEngine.ts
'use client';

import { useState, useCallback, useRef, useEffect } from 'react';
import { type GameState, type StoryNode, type Choice, type Attributes } from '@/types/game';
import { type Chapter } from '@/lib/tokens';
import { useSaveSystem } from './useSaveSystem';

const INITIAL_STATE: GameState = {
  currentNode: 'prologue_1',
  chapter: 'prologue',
  attributes: {
    spirit: 0,
    sense: 15,
    wisdom: 85,
    science: 99,
    money: 5000,
  },
  relationships: {},
  flags: {},
  achievements: [],
  playTime: 0,
};

export function useGameEngine() {
  const [state, setState] = useState<GameState>(INITIAL_STATE);
  const [isAutoPlaying, setIsAutoPlaying] = useState(false);
  const autoPlayTimerRef = useRef<NodeJS.Timeout | null>(null);
  const playTimeTimerRef = useRef<NodeJS.Timeout | null>(null);
  const { save, load } = useSaveSystem();

  // 游戏时间计时器
  useEffect(() => {
    playTimeTimerRef.current = setInterval(() => {
      setState(prev => ({ ...prev, playTime: prev.playTime + 1 }));
    }, 1000);

    return () => {
      if (playTimeTimerRef.current) {
        clearInterval(playTimeTimerRef.current);
      }
    };
  }, []);

  // 加载节点
  const loadNode = useCallback((nodeId: string, node: StoryNode) => {
    setState(prev => ({
      ...prev,
      currentNode: nodeId,
      chapter: node.chapter,
    }));
  }, []);

  // 处理选择
  const makeChoice = useCallback((choice: Choice) => {
    setState(prev => {
      const newState = { ...prev };

      // 应用属性变化
      if (choice.effects?.attributes) {
        const newAttributes = { ...newState.attributes };
        Object.entries(choice.effects.attributes).forEach(([key, value]) => {
          if (value !== undefined) {
            (newAttributes as any)[key] += value;
          }
        });
        newState.attributes = newAttributes;
      }

      // 应用标记
      if (choice.effects?.flags) {
        newState.flags = { ...newState.flags, ...choice.effects.flags };
      }

      // 应用关系变化
      if (choice.effects?.relationships) {
        Object.entries(choice.effects.relationships).forEach(([key, value]) => {
          newState.relationships = {
            ...newState.relationships,
            [key]: (newState.relationships[key] || 0) + value,
          };
        });
      }

      // 解锁成就
      if (choice.effects?.achievements) {
        newState.achievements = [
          ...new Set([...newState.achievements, ...choice.effects.achievements]),
        ];
      }

      return newState;
    });
  }, []);

  // 自动播放
  const startAutoPlay = useCallback((delay: number, callback: () => void) => {
    setIsAutoPlaying(true);
    autoPlayTimerRef.current = setTimeout(() => {
      callback();
      setIsAutoPlaying(false);
    }, delay);
  }, []);

  const stopAutoPlay = useCallback(() => {
    if (autoPlayTimerRef.current) {
      clearTimeout(autoPlayTimerRef.current);
      autoPlayTimerRef.current = null;
    }
    setIsAutoPlaying(false);
  }, []);

  // 存档
  const saveGame = useCallback((slot: string = 'auto') => {
    save(state, slot);
  }, [state, save]);

  // 读档
  const loadGame = useCallback((slot: string = 'auto') => {
    const loadedState = load(slot);
    if (loadedState) {
      setState(loadedState);
      return true;
    }
    return false;
  }, [load]);

  // 重新开始
  const restart = useCallback(() => {
    setState(INITIAL_STATE);
  }, []);

  return {
    state,
    isAutoPlaying,
    loadNode,
    makeChoice,
    startAutoPlay,
    stopAutoPlay,
    saveGame,
    loadGame,
    restart,
  };
}
```

### 4.2 存档系统Hook

```typescript
// src/hooks/useSaveSystem.ts
'use client';

import { useCallback } from 'react';
import { type GameState, type SaveData } from '@/types/game';

const SAVE_PREFIX = 'phd_save_';
const CURRENT_VERSION = '2.0';

export function useSaveSystem() {
  const save = useCallback((state: GameState, slot: string = 'auto') => {
    const saveData: SaveData = {
      state,
      timestamp: Date.now(),
      version: CURRENT_VERSION,
    };
    localStorage.setItem(`${SAVE_PREFIX}${slot}`, JSON.stringify(saveData));
  }, []);

  const load = useCallback((slot: string = 'auto'): GameState | null => {
    const data = localStorage.getItem(`${SAVE_PREFIX}${slot}`);
    if (!data) return null;

    try {
      const saveData: SaveData = JSON.parse(data);
      return saveData.state;
    } catch {
      return null;
    }
  }, []);

  const deleteSave = useCallback((slot: string = 'auto') => {
    localStorage.removeItem(`${SAVE_PREFIX}${slot}`);
  }, []);

  const getAllSaves = useCallback((): Array<{ slot: string; timestamp: number; chapter: string }> => {
    const saves: Array<{ slot: string; timestamp: number; chapter: string }> = [];
    
    for (let i = 0; i < localStorage.length; i++) {
      const key = localStorage.key(i);
      if (key?.startsWith(SAVE_PREFIX)) {
        const slot = key.replace(SAVE_PREFIX, '');
        try {
          const data: SaveData = JSON.parse(localStorage.getItem(key)!);
          saves.push({
            slot,
            timestamp: data.timestamp,
            chapter: data.state.chapter,
          });
        } catch {
          // 忽略损坏的存档
        }
      }
    }

    return saves.sort((a, b) => b.timestamp - a.timestamp);
  }, []);

  const exportSave = useCallback((slot: string = 'auto') => {
    const data = localStorage.getItem(`${SAVE_PREFIX}${slot}`);
    if (!data) return;

    const blob = new Blob([data], { type: 'application/json' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `phd_save_${Date.now()}.json`;
    a.click();
    URL.revokeObjectURL(url);
  }, []);

  const importSave = useCallback(async (file: File): Promise<GameState | null> => {
    return new Promise((resolve) => {
      const reader = new FileReader();
      reader.onload = (e) => {
        try {
          const saveData: SaveData = JSON.parse(e.target?.result as string);
          resolve(saveData.state);
        } catch {
          resolve(null);
        }
      };
      reader.readAsText(file);
    });
  }, []);

  return { save, load, deleteSave, getAllSaves, exportSave, importSave };
}
```

## 11. 痕迹系统（终章防白玩机制）

### 11.1 系统概述

痕迹系统是终章"防白玩"设计的核心技术实现。当玩家从梦境醒来后，系统会在现实中留下三层痕迹（身体、物品、知识），证明梦中的经历是真实的，而非虚构。

### 11.2 痕迹数据结构

```typescript
// 痕迹类型
export type TraceType = 'body' | 'item' | 'knowledge';

// 痕迹状态
export interface Trace {
  type: TraceType;
  id: string;
  name: string;
  description: string;
  discovered: boolean;
}

// 痕迹系统状态
export interface TraceSystem {
  traces: {
    runeMark: Trace;      // 身体痕迹：符文印记
    scar: Trace;          // 身体痕迹：疤痕
    stoneWarmth: Trace;   // 物品痕迹：石头温度
    oceanSound: Trace;    // 物品痕迹：海浪声
    runeMeaning: Trace;   // 知识痕迹：符文含义
  };
  discoveryProgress: number;  // 0-3，三层痕迹
  truthRevealed: boolean;
}
```

### 11.3 痕迹触发机制

```typescript
// 痕迹系统Hook
export function useTraceSystem() {
  const [traces, setTraces] = useState<TraceSystem>(initialTraces);

  // 发现身体痕迹（觉醒时自动触发）
  const discoverBodyTrace = useCallback(() => {
    setTraces(prev => ({
      ...prev,
      traces: {
        ...prev.traces,
        runeMark: { ...prev.traces.runeMark, discovered: true },
        scar: { ...prev.traces.scar, discovered: true },
      },
      discoveryProgress: 1,
    }));
  }, []);

  // 发现物品痕迹（触摸石头时）
  const discoverItemTrace = useCallback(() => {
    setTraces(prev => ({
      ...prev,
      traces: {

        oceanSound: { ...prev.traces.oceanSound, discovered: true },
      },
      discoveryProgress: 2,
    }));
  }, []);

  // 发现知识痕迹（注视符文时）
  const discoverKnowledgeTrace = useCallback(() => {
    setTraces(prev => ({
      ...prev,
      traces: {
        ...prev.traces,
        runeMeaning: { ...prev.traces.runeMeaning, discovered: true },
      },
      discoveryProgress: 3,
      truthRevealed: true,
    }));
  }, []);

  return { traces, discoverBodyTrace, discoverItemTrace, discoverKnowledgeTrace };
}
```

```typescript
export function useMemorySystem() {
```

### 12.3 真实时间机制

**系统概述**：
某些修炼需要真实时间等待，期间屏幕变黑，只能听到海浪声，增加沉浸感和"修炼"的真实感。

**核心实现**：

```typescript
// 真实时间系统Hook
export function useRealTimeSystem() {
  const [waitingTasks, setWaitingTasks] = useState<Map<string, {
    startTime: number;
    endTime: number;
    onComplete: () => void;
    progress: number;
  }>>(new Map());

  // 开始等待任务
  const startWaitingTask = useCallback((taskId: string, duration: number, onComplete: () => void) => {
    const startTime = Date.now();
    const endTime = startTime + duration;

    setWaitingTasks(prev => new Map(prev).set(taskId, {
      startTime,
      endTime,
      onComplete,
      progress: 0,
    }));

    // 更新进度
    const updateProgress = () => {
      setWaitingTasks(prev => {
        const task = prev.get(taskId);
        if (!task) return prev;

        const now = Date.now();
        const progress = Math.min((now - task.startTime) / (task.endTime - task.startTime), 1);

        if (progress >= 1) {
          task.onComplete();
          const newTasks = new Map(prev);
          newTasks.delete(taskId);
          return newTasks;
        }

        return new Map(prev).set(taskId, { ...task, progress });
      });

      requestAnimationFrame(updateProgress);
    };

    requestAnimationFrame(updateProgress);
  }, []);

  return { waitingTasks, startWaitingTask };
}
```

```typescript
// 抉择统计系统
export function useChoiceStatsSystem() {
  const [stats, setStats] = useState<Record<string, number>>(() => {
    const data = localStorage.getItem('phd_choice_stats');
    return data ? JSON.parse(data) : {};
  });

  // 记录选择
  const recordChoice = useCallback((sceneId: string, choiceId: string) => {
    const key = `${sceneId}_${choiceId}`;
    setStats(prev => {
      const newStats = { ...prev, [key]: (prev[key] || 0) + 1 };
      localStorage.setItem('phd_choice_stats', JSON.stringify(newStats));
      return newStats;
    });
  }, []);

  // 获取选择统计
  const getChoiceStats = useCallback((sceneId: string, choiceId: string): number => {
    const key = `${sceneId}_${choiceId}`;
    const total = Object.entries(stats)
      .filter(([k]) => k.startsWith(sceneId))
      .reduce((sum, [, count]) => sum + count, 0);
    const choiceCount = stats[key] || 0;
    return total > 0 ? Math.round((choiceCount / total) * 100) : 0;
  }, [stats]);

  return { stats, recordChoice, getChoiceStats };
}
```
